Send registered-event SMS notification

Endpoint: /gwt.alrajhibank.com.sa/mp-notification-sms/v1/registered-event

Swagger Link

This path is used to send registered event SMS notification.

Request Example

                                        
    Request Body
    {
        "Data": {
          "ClientTimestamp": "2022-02-17T18:25:43.511+03:00",
          "EventCode": "QAEMALOG",
          "RecipientsList": [
            {
              "PreferredLanguage": "EN",
              "NotificationsMethodsInformationList": [
                {
                  "Contact": "966500000111"
                }
              ],
              "ParametersList": [
                {
                  "ParameterCode": "TRN_AMNT",
                  "ParameterValue": "1000"
                }
              ]
            }
          ]
        }
      }

    --header 'Accept-Language: ar' \
    --header 'Content-Type: application/json' \
    --header 'Accept: application/json' \
    --header 'Authorization: ••••••' \
    --data '{
    "Data": {
    "ClientTimestamp": "2022-02-17T18:25:43.511+03:00",
    "EventCode": "QAEMALOG",
    "RecipientsList": [
    {
    "PreferredLanguage": "EN",

    {

    ],
